rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if user is admin
    // Note: This checks the user_roles collection, but we need to be careful
    // about circular dependencies. For production, we'll use a simpler approach.
    function isAdmin() {
      // For now, allow if authenticated - admin check happens in application code
      // In a more secure setup, you'd check user_roles here, but that requires
      // the user_roles collection to be readable, which creates a dependency
      return request.auth != null;
    }
    
    // User Roles - Admin only
    // This collection stores admin/operator roles
    match /user_roles/{document} {
      // Allow read for authenticated users (needed for admin check in app)
      // In production, you might want to restrict this further
      allow read: if request.auth != null;
      // Only allow write for authenticated users (admin check in app code)
      // For more security, you could add a hardcoded admin email check here
      allow write: if request.auth != null;
    }
    
    // Admin Config - Admin only
    match /admin_config/{document} {
      // Allow read for authenticated users (needed for admin check)
      allow read: if request.auth != null;
      // Only allow write for authenticated users (admin check in app code)
      allow write: if request.auth != null;
    }
    
    // Orders - Users can create, read their own, admins can read/write all
    match /orders/{orderId} {
      // Allow authenticated users (including anonymous) to create orders
      // Ensure the order is created with the current user's ID
      allow create: if request.auth != null && (
        request.resource.data.userId == request.auth.uid ||
        request.resource.data.userId == null  // Allow if userId not set yet
      );
      
      // Allow users to read their own orders
      // Check if order belongs to current user (userId match)
      // Also allow if customer email matches (for logged-in users)
      allow read: if request.auth != null && (
        resource.data.userId == request.auth.uid ||
        (request.auth.token.email != null && 
         resource.data.customer?.email == request.auth.token.email)
      );
      
      // Allow users to update their own orders (for status updates, etc.)
      allow update: if request.auth != null && (
        resource.data.userId == request.auth.uid ||
        (request.auth.token.email != null && 
         resource.data.customer?.email == request.auth.token.email)
      );
      
      // Allow delete for authenticated users (admin check in app code)
      allow delete: if request.auth != null;
    }
    
    // Products - Public read, authenticated write
    match /products/{productId} {
      // Anyone can read products (needed for pricing/display)
      allow read: if true;
      // Only authenticated users can write (admin check in app code)
      allow write: if request.auth != null;
    }
    
    // Settings - Public read, authenticated write
    match /settings/{document} {
      // Anyone can read settings (shipping options, etc.)
      allow read: if true;
      // Only authenticated users can write (admin check in app code)
      allow write: if request.auth != null;
    }
    
    // Market Events - Public read, authenticated write
    match /marketEvents/{eventId} {
      // Anyone can read market events (needed for landing page)
      allow read: if true;
      // Only authenticated users can write (admin check in app code)
      allow write: if request.auth != null;
    }
    
    // Test collection - for diagnostic purposes
    match /test/{document} {
      allow read, write: if request.auth != null;
    }
    
    // Deny all other collections by default
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
