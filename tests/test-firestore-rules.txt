rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Products: Allow read for everyone, write for authenticated users
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Market Events: Allow read for everyone, write for authenticated users
    match /marketEvents/{eventId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Admin Config: Allow read for everyone (needed for admin check), write for authenticated users
    match /admin_config/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Settings: Allow read for everyone, write for authenticated users
    match /settings/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Orders: Allow read/write for authenticated users (including anonymous)
    match /orders/{orderId} {
      allow read, write: if request.auth != null;
    }

    // Themes collection: Allow read for everyone, write for authenticated users
    match /themes/{themeId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Allow all other collections (for development/testing)
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
